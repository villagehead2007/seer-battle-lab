'use strict';const{test}=require('node:test'),assert=require('node:assert/strict'),fs=require('fs'),api=require('../seerapi.cjs'),r=require('../research/4169-source-check.json').records,raw=r.find(x=>x.url.endsWith('skill/13983')).data,pet=r.find(x=>x.url.endsWith('pet/4169')).data,learn=pet.skill.find(x=>x.skill.id===13983);
test('4169真实13983三目标来源单挑正常伤害且其余目标延期，错误hash/数量/其他ID不扩大兼容',()=>{assert.equal(raw.hash,'5e7a921e');assert.equal(raw.atk_num,3);const before=JSON.stringify(raw),m=api.convertMove(raw,learn,'普通',{deferTeamEffects:true});assert.ok(!m.limitations.includes('多目标攻击尚未适配'));assert.ok(m.deferredEffects.some(x=>x.field==='atk_num'&&x.description.includes('atk_num=3')));for(const patch of [{hash:'wrong'},{atk_num:2},{atk_num:'3'},{id:999993}])assert.ok(api.convertMove({...raw,...patch},learn,'普通',{deferTeamEffects:true}).limitations.includes('多目标攻击尚未适配'));assert.ok(api.convertMove(raw,learn,'普通',{deferTeamEffects:false}).limitations.includes('多目标攻击尚未适配'));assert.equal(JSON.stringify(raw),before);const h=require('./helpers/4169-battle.cjs'),s=h.make({moveId:13983,index:null},{omit:true}),o=h.turn(s);assert.equal(o.damageEvents.filter(x=>x.type==='attack'&&x.targetSide===1).length,1);assert.ok(o.fighters[1].hp<9000);assert.equal(o.fighters[0].pp[h.key(13983)],9);});
test('13983新增支持保留既有真实11544来源的精确hash和两目标约束',()=>{const old=require('../research/3907-source-check.json').records.find(x=>x.url==='https://api.seerapi.com/v1/skill/11544');assert.equal(old.url,'https://api.seerapi.com/v1/skill/11544');assert.equal(old.data.hash,'6070eb31');assert.equal(old.data.atk_num,2);const l={learning_level:1,is_fifth:false,is_special:false};assert.ok(!api.convertMove(old.data,l,'普通',{deferTeamEffects:true}).limitations.includes('多目标攻击尚未适配'));assert.ok(api.convertMove({...old.data,atk_num:3},l,'普通',{deferTeamEffects:true}).limitations.includes('多目标攻击尚未适配'));});
