'use strict';
const {test}=require('node:test'),assert=require('node:assert/strict');const {E,source,make,turn,pick,timer,key}=require('./helpers/3886-battle.cjs');
function incoming(s){return s.damageEvents.find(e=>e.type==='attack'&&e.targetSide===0)?.settledDamage??0;}
function plain(s){const c=structuredClone(s);c.fighters[0].soulId=null;return turn(c,900031,900040);}
test('3886原魂印按护盾后同一伤害判断高低分支，减免不二次反弹，半血等值不触发',()=>{for(const spa of [180,9000]){const s=make(undefined,{soul:1155,aStats:{maxHp:1000},bStats:{spa}}),raw=incoming(plain(s)),o=turn(s,900031,900040);if(raw>500){assert.equal(incoming(o),Math.floor(raw*.4));assert.equal(o.fighters[1].hp,9000);}else{assert.equal(incoming(o),raw);assert.equal(o.fighters[1].hp,9000-raw);}const equal=make(undefined,{soul:1155,aStats:{maxHp:1000},bStats:{spa}});equal.fighters[0].stats.maxHp=raw*2;equal.fighters[0].hp=raw*2;const e=turn(equal,900031,900040);assert.equal(incoming(e),raw);assert.equal(e.fighters[1].hp,9000);}let s=make(undefined,{soul:1155});s.fighters[0].suitState.shield=99999;let o=turn(s,900031,900040);assert.equal(incoming(o),0);assert.equal(o.fighters[1].hp,9000);s=make(undefined,{soul:1155,hp:1});const raw=incoming(plain(s));o=turn(s,900031,900040);assert.equal(o.fighters[0].hp,0);assert.equal(incoming(o),raw);assert.equal(o.fighters[1].hp,9000-raw);});
test('3886原魂印无视强化、冰封未触发才授予下次200限伤，免疫不绕过',()=>{let s=make(undefined,{soul:1155});s.fighters[1].stages.def=2;s.fighters[1].counts.immune=1;let o=turn(s);assert.equal(o.fighters[1].stages.def,2);assert.equal(o.fighters[1].ailments['冰封'],undefined);assert.equal(o.fighters[0].soulState.coldNextCap,true);o.moveDefinitions[key(900040)].power=9999;o=turn(o,900031,900040);assert.ok(incoming(o)<=200);assert.equal(o.fighters[0].soulState.coldNextCap,undefined);s=make(undefined,{soul:1155});s.fighters[1].stages.atk=1;o=turn(s);assert.ok(o.fighters[1].ailments['冰封']);assert.equal(o.fighters[0].soulState.coldNextCap,undefined);});
test('3886觉醒按回合累计减伤后低额双向真伤，反馈不递归累计，真伤增加三道上限',()=>{const s=make(undefined,{soul:1971}),raw=incoming(plain(s)),saved=raw-Math.floor(raw*.4),o=turn(s,900031,900040);assert.equal(incoming(o),Math.floor(raw*.4));assert.equal(o.fighters[0].hp,9000-raw);assert.equal(o.fighters[1].hp,9000-saved);assert.equal(o.fighters[0].soulState.coldEssence,1);const trueHits=o.damageEvents.filter(e=>e.type==='true');assert.equal(trueHits.length,2);assert.ok(trueHits.every(e=>e.settledDamage===saved));assert.equal(o.fighters[0].soulState.coldSaved,0);});
test('3886觉醒双向反馈将同时死亡时敌方保留1HP，双方完整结算伤害不截断',()=>{let s=make(undefined,{soul:1971});const raw=incoming(plain(s)),left=Math.floor(raw*.4),saved=raw-left;s.fighters[0].hp=left+1;s.fighters[1].hp=saved;const o=turn(s,900031,900040);assert.equal(o.fighters[0].hp,0);assert.equal(o.fighters[1].hp,1);const hits=o.damageEvents.filter(e=>e.type==='true');assert.equal(hits.length,2);assert.ok(hits.every(e=>e.settledDamage===saved));assert.equal(hits[1].hpLoss,saved-1);});
test('3886觉醒多段累计高于200才冻伤，不逐段触发低额反馈',()=>{const s=make(undefined,{soul:1971,opponentMove:37809,opponentEffectId:2268,bStats:{atk:2000}});s.fighters[0].soulState.coldEssence=1;const o=turn(s,900031,900042);assert.ok(o.damageEvents.filter(e=>e.type==='attack').length>1);assert.ok(o.fighters[1].ailments['冻伤']>=3);assert.equal(o.damageEvents.filter(e=>e.type==='true').length,0);});
test('3886觉醒魂魄和异常决定忽略先制阈值，零魂魄不忽略',()=>{for(const n of [0,1,3]){const s=make(undefined,{soul:1971,aStats:{spe:900},bStats:{spe:100}});s.fighters[0].soulState.coldEssence=n;s.fighters[0].ailments['凝滞']=3;timer(s,1,'id984',{amount:3});const o=turn(s,900031,900040);assert.equal(o.turns[0].first,n>0?0:1);}});
test('3886觉醒冰封成功与免疫两分支、零魂魄主动附加肃霜，技能限伤后按魂数扣点',()=>{for(const n of [0,1,3])for(const immune of [false,true]){const s=make(undefined,{soul:1971});s.fighters[0].soulState.coldEssence=n;s.fighters[1].stages.def=1;if(immune)s.fighters[1].counts.immune=1;const o=turn(s);assert.equal(o.fighters[1].soulState.coldBan??0,n===0||immune?200:0);}const s=make(undefined,{soul:1971});s.fighters[0].soulState.coldEssence=3;s.fighters[1].soulState.coldBan=200;s.moveDefinitions[key(900040)].power=9999;const o=turn(s,900031,900040);assert.equal(incoming(o),80);assert.equal(o.fighters[1].soulState.coldBan,150);});
test('3886觉醒2265按1/2/3道分级无视弱化护盾和限伤，原负等级保持',()=>{for(const n of [0,1,2,3]){const s=make(pick(2265)[0]);s.fighters[0].soulState.coldEssence=n;s.fighters[0].stages.atk=-2;s.fighters[1].suitState.shield=10000;timer(s,1,'伤害上限',{amount:1});const o=turn(s),loss=9000-o.fighters[1].hp;assert.equal(o.fighters[0].stages.atk,-2);if(n<2)assert.equal(loss,0);else if(n===2)assert.equal(loss,1);else assert.ok(loss>1);}});
