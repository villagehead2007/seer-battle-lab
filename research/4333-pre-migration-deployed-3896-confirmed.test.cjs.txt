'use strict';const {test}=require('node:test'),assert=require('node:assert/strict');const {E,source,make,turn,key,pick,timer}=require('./helpers/3896-battle.cjs');
test('3896原魂印首回合暴击与免弱整回合有效，次回合失效，非暴击正常扣血',()=>{let s=make(undefined,{soul:1165,opponentMove:20513,opponentEffectId:5});let o=turn(s,900031,900042);assert.equal(o.fighters[0].stages.spe,0);o=turn(o,900031,900042);assert.equal(o.fighters[0].stages.spe,-2);s=make(undefined,{soul:1165});timer(s,1,'必定暴击');o=turn(s,900031,900040);assert.equal(o.fighters[0].hp,9000);o=turn(o,900031,900040);assert.ok(o.fighters[0].hp<9000);assert.ok(turn(make(undefined,{soul:1165}),900031,900040).fighters[0].hp<9000);});
test('3896原魂印回合初低血固定差值锁定及攻击先制，属性不附加，高血MISS仍害怕或清回合',()=>{let s=make(undefined,{soul:1165,hp:8000}),o=turn(s);assert.ok(o.damageEvents.some(e=>e.type==='fixed'&&e.settledDamage===1000));assert.equal(o.fighters[0].soulState.gammaDifference,1000);assert.equal((turn(s,900031).damageEvents??[]).filter(e=>e.type==='fixed').length,0);s=make(undefined,{soul:1165,targetHp:8000,hit:false});s.moveDefinitions[key(s.auditMoveId)].accuracy=0;timer(s,1,'doubleDamage');o=turn(s);assert.ok(o.fighters[1].ailments['害怕']||!o.fighters[1].effects.doubleDamage);assert.equal((o.damageEvents??[]).filter(e=>e.type==='attack').length,0);});
test('3896强化首次暴击抵挡后解异常并超频，免弱结束；非暴击不消耗抵挡',()=>{let s=make(undefined,{soul:1959});s.fighters[0].ailments['中毒']=3;timer(s,1,'必定暴击');let o=turn(s,900031,900040);assert.equal(o.fighters[0].hp,9000);assert.equal(o.fighters[0].ailments['中毒'],undefined);assert.equal(o.fighters[0].ailments['超频'],3);assert.equal(o.fighters[0].soulState.gammaCritReady,false);o=turn(o,900031,900040);assert.ok(o.fighters[0].hp<9000);s=make(undefined,{soul:1959});o=turn(s,900031,900040);assert.equal(o.fighters[0].soulState.gammaCritReady,true);});
test('3896强化超频逐次保命不截断完整伤害，1HP仍有伤害事件并清其他异常',()=>{const s=make(undefined,{soul:1959,hp:200,bStats:{spa:9000}});s.fighters[0].soulState.gammaEntered=true;s.fighters[0].soulState.gammaCritReady=false;s.fighters[0].ailments['超频']=3;s.fighters[0].ailments['中毒']=3;let o=turn(s,900031,900040),hit=o.damageEvents.find(e=>e.type==='attack');assert.ok(hit.settledDamage>200);assert.equal(hit.hpLoss,199);assert.equal(o.fighters[0].hp,1);assert.equal(o.fighters[0].ailments['中毒'],undefined);assert.equal(o.fighters[0].ailments['超频'],3);o=turn(o,900031,900040);hit=(o.damageEvents??[]).filter(e=>e.type==='attack').at(-1);assert.equal(hit.hpLoss,0);assert.ok(hit.settledDamage>0);assert.equal(o.fighters[0].hp,1);});
test('3896强化无法切换低血高血等血均重置并转移清除异常，包含超频且免疫仍清自身',()=>{for(const hp of [8000,9000])for(const immune of [false,true]){const s=make(undefined,{soul:1959,hp});s.fighters[0].soulState.gammaEntered=true;s.fighters[0].soulState.gammaCritReady=false;s.fighters[0].ailments['凝滞']=3;s.fighters[0].ailments['超频']=3;if(immune)s.fighters[1].counts.immune=2;const o=turn(s,900031);assert.equal(o.fighters[0].soulState.gammaCritReady,true);assert.equal(o.fighters[0].ailments['凝滞'],undefined);assert.equal(o.fighters[0].ailments['超频'],undefined);assert.equal(!!o.fighters[1].ailments['凝滞'],!immune);assert.equal(!!o.fighters[1].ailments['超频'],!immune);}const s=make(undefined,{soul:1959,targetHp:8000});s.fighters[0].soulState.gammaEntered=true;s.fighters[0].soulState.gammaCritReady=false;timer(s,0,'不能主动切换');assert.equal(turn(s,900031).fighters[0].soulState.gammaCritReady,true);});
test('3896强化低血使用前附加当前差70%真伤，即使MISS，满额伤害后高血恐惧免疫则清回合',()=>{let s=make(undefined,{soul:1959,hp:8000,hit:false});s.moveDefinitions[key(s.auditMoveId)].accuracy=0;let o=turn(s);assert.equal(o.fighters[1].hp,8300);assert.equal(o.damageEvents.find(e=>e.type==='true').settledDamage,700);s=make(undefined,{soul:1959,targetHp:8000});s.fighters[1].counts.immune=1;timer(s,1,'doubleDamage');o=turn(s);assert.equal(o.fighters[1].ailments['害怕'],undefined);assert.equal(o.fighters[1].effects.doubleDamage,undefined);});
test('3896效果1784只抵挡下一次非暴击伤害，暴击不消费，限当回合',()=>{let s=make(pick(1784)[0]);let o=turn(s,undefined,900040);assert.equal(o.fighters[0].hp,9000);s=make(pick(1784)[0]);timer(s,1,'必定暴击');o=turn(s,undefined,900040);assert.ok(o.fighters[0].hp<9000);const hp=o.fighters[0].hp;o=turn(o,900031,900040);assert.ok(o.fighters[0].hp<hp);assert.equal(o.fighters[0].effects['下次非暴击免疫'],undefined);});
test('3896固定数值1241与1214理论回复，不受目标低HP或固定伤害全免影响，触发条件不放宽',()=>{for(const id of [1241,1214])for(const immune of [false,true]){const s=make(pick(id)[0],{hp:1000,targetHp:1});timer(s,1,'攻击伤害免疫');if(id===1214)s.fighters[1].ailments['凝滞']=3;if(immune)s.fighters[1].build.resistance.fixed=100;const o=turn(s);assert.equal(o.fighters[0].hp,id===1241?1400:1200);assert.equal(o.fighters[1].hp,immune?1:0);}const s=make(pick(1214)[0],{hp:1000});assert.equal(turn(s).fighters[0].hp,1000);});
