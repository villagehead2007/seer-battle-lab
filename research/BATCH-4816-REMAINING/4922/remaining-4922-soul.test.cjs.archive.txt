'use strict';const test=require('node:test'),assert=require('node:assert/strict'),H=require('./helpers/remaining-4922.cjs'),{E,make,turn,timer,sum}=H;
test('4922真实2118回合开始非满体才本回红伤加50%，回末耗血不倒推当前增伤',()=>{for(const hp of[9000,8000]){const a=make({move:10001,soul:true,hp}),b=make({move:10001,hp});turn(a,10001,20309);turn(b,10001,20309);if(hp===9000)assert.equal(sum(a),sum(b));else assert.ok(Math.abs(sum(a)-sum(b)*1.5)<=1);assert.equal(a.fighters[0].hp,hp*.8);}const s=make({move:10001,soul:true});turn(s,10001,20309);assert.equal(s.fighters[0].hp,7200);turn(s,10001,20309);assert.equal(s.fighters[0].soulState.ruinDamage2118,2);});
test('4922真实2118消耗当前HP20%直接损失，严格耗后半体边界，低体魂印回满且不拿当前精灵当场下回复',()=>{for(const hp of[9000,6000,5625,5624,5623,1000,1]){const s=make({move:10001,soul:true,hp});turn(s,10001,20309);const cost=Math.floor(hp*.2),left=hp-cost;assert.equal(s.fighters[0].hp,left<4500?9000:left);assert.deepEqual(s.fighters[0].soulState.ruinBench2118,[{round:1,amount:cost,sourceHash:'ef40c509',requiresRealBench:true}]);const ev=(s.damageEvents??[]).filter(e=>e.type==='direct'&&e.sourceSide===0&&e.targetSide===0);assert.equal(ev.length,0);if(cost)assert.ok(s.turns.at(-1).events.some(e=>e.text.includes('直接损失 '+cost+' 点体力')));}});
test('4922真实2118技能禁疗不挡魂印回血，回复倍率仍作用；固粉免疫不阻直接消耗，死亡不执行末回复',()=>{for(const mode of['ban','cut','immune']){const s=make({move:10001,soul:true,hp:1000});if(mode==='ban')timer(s,0,'技能禁疗');if(mode==='cut')timer(s,0,'回复效果削减',{fraction:.5});if(mode==='immune')timer(s,0,'固定百分比免疫');turn(s,10001,20309);assert.equal(s.fighters[0].hp,mode==='cut'?5300:9000);assert.equal(s.fighters[0].soulState.ruinBench2118[0].amount,200);}const p=make({move:10001,soul:true,hp:1,aStats:{def:1},bStats:{atk:9999}});turn(p,10001,10127);assert.equal(p.fighters[0].hp,0);assert.equal(p.fighters[0].soulState.ruinBench2118,undefined);});
