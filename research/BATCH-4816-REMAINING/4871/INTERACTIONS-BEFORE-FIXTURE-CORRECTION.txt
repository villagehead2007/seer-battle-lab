'use strict';
const test=require('node:test'),assert=require('node:assert/strict'),H=require('./helpers/remaining-4871.cjs'),{E,source,key,make,turn,timer,sum}=H;
const row=id=>source.moves.flatMap(m=>m.effectInstances.map((e,index)=>({move:m.id,index,id:e.id}))).find(e=>e.id===id);
function observe(fn,run){const old=E.SOULS;E.SOULS={...old,run(e,c){old.run(e,c);fn(e,c);}};try{return run();}finally{E.SOULS=old;}}
test('4871/1491天敌先制快照为真实双方精灵属性；843后两回合加2并到期，743仅成功下回合加1',()=>{for(const hostile of[false,true]){const s=make(row(1491));s.fighters[0].type='火';s.fighters[1].type=hostile?'水':'草';turn(s);assert.equal(s.fighters[0].soulState.resolvedPriority.value,hostile?1:0);}for(const [id,n]of[[843,2],[743,1]]){const s=make(row(id));turn(s);for(let r=2;r<=(id===843?4:3);r++){turn(s,34462);assert.equal(s.fighters[0].soulState.resolvedPriority.value,r<=(id===843?3:2)?n:0);}}});
test('4871/681后两回合真实必暴且攻击必失被必中覆盖，第三回合恢复MISS；663属性/药剂不触发、攻击MISS仍睡眠',()=>{const s=make(row(681));turn(s);delete s.fighters[0].effects['攻击必中'];timer(s,0,'攻击必失');for(let r=2;r<=4;r++){let crit;observe((e,c)=>{if(e==='afterAttack'&&c.a.side===0)crit=c.crit;},()=>turn(s,34462));if(r<=3){assert.ok(sum(s)>0);assert.equal(crit,true);}else assert.equal(sum(s),0);}for(const m of[20117,null,10127]){const a=make(row(663));timer(a,1,'攻击必失');turn(a,29265,m);assert.equal(!!a.fighters[1].ailments['睡眠'],m===10127);}});
test('4871/781属性封锁跨两回合且保护不授，463红伤额度逐回合重置到期后恢复',()=>{for(const protect of[false,true]){const s=make(row(781));if(protect)timer(s,1,'protect');turn(s);assert.equal(s.fighters[1].stages.atk,protect?1:0);for(let r=2;r<=3;r++){s.fighters[1].stages.atk=0;turn(s,34462);assert.equal(s.fighters[1].stages.atk,protect||r===3?1:0);}}const s=make(row(463));turn(s);for(let r=2;r<=4;r++){let damage;observe((e,c)=>{if(e==='beforeUse'&&c.a.side===1)damage=c.hurt(c.t,400,'额度重置与到期','attack');},()=>turn(s,34462));assert.equal(damage,r<=3?0:400);}});
test('4871/810当前封命中效果不撤销此前永久降威，29已命中零红仍固定、MISS或整招无效不附加',()=>{const s=make(row(810));turn(s);s.fighters[0].pp[key(38272)]=1;timer(s,0,'攻击技能命中效果失效');turn(s);assert.equal(sum(s),0);assert.equal(Object.values(s.fighters[0].soulState.hits8104871)[0],1);for(const mode of['zero','miss','invalid']){const a=make(row(29));timer(a,mode==='zero'?1:0,mode==='zero'?'攻击伤害免疫':mode==='miss'?'攻击必失':'skillBlock');turn(a);assert.equal(sum(a,'fixed'),mode==='zero'?1000:0);}});
