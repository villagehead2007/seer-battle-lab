'use strict';const test=require('node:test'),assert=require('node:assert/strict'),H=require('./helpers/remaining-4911.cjs'),{E,make,turn,timer,key,sum,keys}=H;
function observe(fn,run){const old=E.SOULS;E.SOULS={...old,run(e,c){old.run(e,c);fn(e,c);}};try{return run();}finally{E.SOULS=old;}}
test('4911真实2097首次双方骑士决斗，双方下次实际使用技能转化真实第五但PP沿原选择槽、排序采用第五先制',()=>{const s=make({move:10001,soul:true}),seen=[];assert.ok(s.fighters.every(f=>f.soulState.knightDuel4911.kind==='duel'));observe((e,c)=>{if(e==='beforeUse')seen.push([c.a.side,c.m.wikiId,c.m.id,c.m.duelFifth4911]);},()=>turn(s,10001,20309));assert.deepEqual(seen.map(x=>x.slice(0,2)),[[1,37911],[0,38419]]);assert.equal(s.fighters[0].pp[key(10001)],34);assert.equal(s.fighters[0].pp[key(38419)],5);assert.equal(s.fighters[1].pp[key(20309)],9);assert.equal(s.fighters[1].pp[key(37911)],5);assert.ok(s.fighters.every(f=>!f.soulState.knightDuel4911.transformNext));assert.ok(s.fighters[1].ailments['冰封']);assert.ok(sum(s)>0);});
test('4911真实2097决斗进入清异常公共状态，已有任一骑士类状态时不重新附双方，未覆盖已有期限',()=>{const s=make({move:10001,soul:true});s.fighters[0].ailments['烧伤']=3;s.fighters[1].ailments['中毒']=4;delete s.fighters[0].soulState.knightDuel4911;delete s.fighters[1].soulState.knightDuel4911;E.SOULS.registry[2097].abilities.enter({a:s.fighters[0],t:s.fighters[1],round:1});assert.equal(s.fighters[0].ailments['烧伤'],undefined);assert.equal(s.fighters[1].ailments['中毒'],undefined);s.fighters[0].soulState.knightDuel4911={kind:'defeat',end:4};delete s.fighters[1].soulState.knightDuel4911;E.SOULS.registry[2097].abilities.enter({a:s.fighters[0],t:s.fighters[1],round:2});assert.equal(s.fighters[0].soulState.knightDuel4911.end,4);assert.equal(s.fighters[1].soulState.knightDuel4911,undefined);});
test('4911真实2097决斗控制未用与药剂保留下次转化，整招无效实际使用消费；无可选第五阻止技能且不扣原槽PP',()=>{for(const mode of['control','potion','invalid','zeroFifth']){const s=make({move:10001,soul:true});if(mode==='control')s.fighters[0].ailments['害怕']=1;if(mode==='invalid')timer(s,0,'skillBlock');if(mode==='zeroFifth')s.fighters[0].pp[key(38419)]=0;turn(s,mode==='potion'?null:10001,20309);assert.equal(s.fighters[0].soulState.knightDuel4911.transformNext,mode!=='invalid');assert.equal(s.fighters[0].pp[key(10001)],mode==='invalid'?34:35);if(mode==='zeroFifth')assert.equal(s.fighters[0].pp[key(38419)],0);}});
