const fs=require('fs'),cp=require('child_process'),crypto=require('crypto'),[id,label,...files]=process.argv.slice(2),dir=__dirname+'/'+id,sha=f=>crypto.createHash('sha256').update(fs.readFileSync(f)).digest('hex'),inputs=['completion-policy.cjs','dist/range-effects.js','dist/effect-registry.js','dist/souls.js','dist/range-souls.js','local-species.cjs','dist/status-rules.js','dist/soul-registry.js','dist/characteristics.js','dist/equipment.js','dist/app.js','dist/pp-rules.js','tests/helpers/remaining-battle.cjs',...files],hashes=Object.fromEntries(inputs.map(f=>[f,sha(f)])),at=new Date().toISOString(),log=dir+'/'+label+'.log',fd=fs.openSync(log,'w');
