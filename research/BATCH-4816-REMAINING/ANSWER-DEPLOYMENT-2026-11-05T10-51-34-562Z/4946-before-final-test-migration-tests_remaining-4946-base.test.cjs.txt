'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{source,make,turn,key}=require('./helpers/remaining-4946.cjs');test('4946全部21招基础行为独立于待答效果验证：真实PP扣1、0PP拒绝且回合与HP不变',()=>{for(const m of source.moves){const s=make({move:m.id,index:-1}),before=s.fighters[0].pp[key(m.id)];assert.equal(before,m.pp);turn(s,m.id,20309);assert.equal(s.fighters[0].pp[key(m.id)],before-1);assert.equal(s.round,2);assert.ok(s.fighters.every(f=>Number.isFinite(f.hp)&&f.hp>0&&f.hp<=f.stats.maxHp));const p=make({move:m.id,index:-1});p.fighters[0].pp[key(m.id)]=0;const hp=p.fighters.map(f=>f.hp);assert.throws(()=>turn(p,m.id,20309),/有 PP/);assert.equal(p.round,1);assert.deepEqual(p.fighters.map(f=>f.hp),hp);assert.equal(p.fighters[0].pp[key(m.id)],0);}});
