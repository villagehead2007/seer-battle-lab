'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{source,make,turn,key}=require('./helpers/remaining-4912.cjs');
test('4912全部17招各64种子真实HP和PP，原生2099全部17招首轮执行与火系消费',()=>{for(const m of source.moves)for(let seed=1;seed<=64;seed++){const s=make({move:m.id,seed:Math.imul(seed,2654435761)>>>0}),pp=s.fighters[0].pp[key(m.id)];turn(s,m.id,20309);assert.equal(s.round,2);assert.equal(s.fighters[0].pp[key(m.id)],pp-1);assert.ok(s.fighters.every(f=>Number.isFinite(f.hp)&&f.hp>=0&&f.hp<=f.stats.maxHp));}for(const m of source.moves){const s=make({move:m.id,soul:true}),pp=s.fighters[0].pp[key(m.id)];turn(s,m.id,20309);assert.equal(s.round,2);assert.equal(s.fighters[0].pp[key(m.id)],pp-1);assert.equal(s.fighters[0].soulState.moth2099.ownFirst,['physical','special'].includes(m.kind));assert.equal(s.fighters[0].type,m.fifth?'虫':source.type);assert.ok(s.fighters.every(f=>Number.isFinite(f.hp)));}});
