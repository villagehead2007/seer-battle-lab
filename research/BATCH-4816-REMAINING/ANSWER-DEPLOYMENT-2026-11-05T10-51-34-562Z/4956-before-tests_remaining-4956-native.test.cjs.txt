'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,make,turn,key,source}=require('./helpers/remaining-4956.cjs');
test('4956全部17招基础真实分类HP和源PP，零威力攻击保留零红事件，原先制与0PP拒绝',()=>{for(const m of source.moves){const s=make({move:m.id,index:-1,aStats:{atk:180,spa:18}});turn(s,m.id,20117);assert.equal(s.fighters[0].pp[key(m.id)],m.pp-1);assert.equal(s.fighters[0].soulState.resolvedPriority.value,m.priority);const es=(s.damageEvents??[]).filter(e=>e.sourceSide===0&&e.type==='attack');assert.equal(es.length,m.kind==='status'?0:1,m.name);if(es.length){assert.equal(es[0].attackValue,m.kind==='physical'?180:18);assert.equal(s.fighters[1].hp,9000-es[0].hpLoss);assert.equal(es[0].settledDamage>0,m.power>0);}const z=make({move:m.id,index:-1});z.fighters[0].pp[key(m.id)]=0;assert.throws(()=>turn(z,m.id,20117),/有 PP/);}});
test('4956全部17招真实有序效果与2153原链仅反馈未知分支显式隔离，其余当前已知行为真实HP和PP',()=>{const old=E.SOULS;E.SOULS={...old,run(e,c){if(e==='combatEnd'&&c.a.side===0&&c.a.soulId===2153&&Object.values(c.a.ailments).some(end=>end>=c.round))return;old.run(e,c);}};try{for(const m of source.moves)for(let seed=1;seed<=4;seed++){const s=make({move:m.id,soul:true,seed:seed*987654321});turn(s,m.id,20117);assert.equal(s.fighters[0].pp[key(m.id)],m.pp-1,m.name);assert.ok(s.fighters.every(f=>f.hp>=0&&f.hp<=f.stats.maxHp));}}finally{E.SOULS=old;}});
