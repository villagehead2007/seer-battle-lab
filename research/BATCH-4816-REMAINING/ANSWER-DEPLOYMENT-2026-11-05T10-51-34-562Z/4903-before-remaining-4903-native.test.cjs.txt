'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{source,make,turn,key}=require('./helpers/remaining-4903.cjs');
test('4903全部26招各64种子基础实际PP/HP与2090首次出战26招原生明确分支，不冒称待答交互完成',()=>{for(const m of source.moves)for(let seed=1;seed<=64;seed++){const s=make({move:m.id,seed:Math.imul(seed,2654435761)>>>0}),before=s.fighters[0].pp[key(m.id)];turn(s,m.id,20117);assert.equal(s.round,2);assert.ok(Number.isFinite(s.fighters[0].hp)&&Number.isFinite(s.fighters[1].hp));assert.ok(s.fighters[0].pp[key(m.id)]<=before);assert.ok(s.fighters.every(f=>f.hp>=0&&f.hp<=f.stats.maxHp));}for(const m of source.moves){const s=make({move:m.id,soul:true});turn(s,m.id,20117);assert.equal(s.round,2);assert.equal(s.fighters[0].soulState.emperor4903.entries,1);assert.equal(s.fighters[0].soulState.emperor4903.entryTriggers,1);assert.ok(Number.isFinite(s.fighters[0].hp)&&Number.isFinite(s.fighters[1].hp));}});
