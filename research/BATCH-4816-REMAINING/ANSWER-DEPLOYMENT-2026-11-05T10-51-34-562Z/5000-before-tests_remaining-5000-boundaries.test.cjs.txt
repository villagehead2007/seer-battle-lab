'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,make,turn,timer,key,source}=require('./helpers/remaining-5000.cjs');
test('5000/753轮回真实发生过强化、消耗和PP回复后回末还原选择快照，免弱不阻返回原等级',()=>{const s=make({move:25679,soul:true,hp:100,aStats:{spe:1},bStats:{atk:9999,spe:500}});for(const k of s.fighters[0].moves)s.fighters[0].pp[k]=1;const before={hp:100,pp:{...s.fighters[0].pp},stages:{...s.fighters[0].stages}},old=E.SOULS;let actualStages;E.SOULS={...old,run(e,c){old.run(e,c);if(e==='afterUse'&&c.a.side===0)actualStages={...c.a.stages};}};try{turn(s,25679,10127);}finally{E.SOULS=old;}assert.ok(actualStages.atk>0);assert.ok(s.fighters[0].soulState.pony753.lifePP[0].chosen.some(x=>x.before!==x.after));assert.equal(s.fighters[0].hp,before.hp);assert.deepEqual(s.fighters[0].pp,before.pp);assert.deepEqual(s.fighters[0].stages,before.stages);});
