'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{make,turn,key}=require('./helpers/remaining-5000.cjs');
const red=s=>(s.damageEvents??[]).filter(e=>e.round===s.round-1&&e.type==='attack'&&e.sourceSide===1).reduce((n,e)=>n+e.settledDamage,0);
const setup=()=>make({move:20498,loadout:[20498,10993,10996,20606,31143],enemyLoadout:[37920,20117,10127,20309,37911]});
test('5000/52持续两回合逐轮判断实际先手攻击，施放时后手仍在次轮生效，后手攻击或控制未用不封对手，到期停止',()=>{for(const mode of['first','last','control']){const s=setup();turn(s,20498,37920);assert.equal(s.turns[0].first,1);if(mode==='last'){s.fighters[0].stats.spe=1;s.fighters[1].stats.spe=500;}if(mode==='control')s.fighters[0].ailments['疲惫']=2;turn(s,10993,10127);assert.equal(red(s)===0,mode==='first');if(mode==='first'){turn(s,10993,10127);assert.ok(red(s)>0);}}});
