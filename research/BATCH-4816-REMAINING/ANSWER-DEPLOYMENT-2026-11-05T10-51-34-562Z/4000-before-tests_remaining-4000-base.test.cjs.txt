'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,make,turn,key,source}=require('./helpers/remaining-4000.cjs');
test('4000全部17招基础组件以不同双攻面板验证真实分类、源威力公式、HP变化及PP，明确隔离另行验证的效果指令',()=>{for(const m of source.moves){const s=make({move:m.id,index:-1,aStats:{atk:180,spa:18}});turn(s,m.id,20117);assert.equal(s.fighters[0].pp[key(m.id)],m.pp-1,m.name);const events=(s.damageEvents??[]).filter(e=>e.sourceSide===0&&e.targetSide===1&&e.type==='attack');if(m.kind==='status'){assert.equal(events.length,0);assert.equal(s.fighters[1].hp,9000);continue;}assert.equal(events.length,1);const event=events[0],attack=m.kind==='physical'?180:18,defense=m.kind==='physical'?s.fighters[1].stats.def:s.fighters[1].stats.spd;assert.equal(event.attackValue,attack,m.name);const log=s.turns[0].events.find(e=>e.damage?.sourceSide===0&&e.damage?.type==='attack'),roll=Number(log.detail.match(/浮动 (\d+)%/)[1]),base=Math.floor(Math.floor((2*s.fighters[0].level/5+2)*m.power*attack/defense)/50)+2,affinity=E.typeEffectiveness(m.type,s.fighters[1].type,s.typeChart),expected=affinity===0?0:Math.max(1,Math.floor(base*E.sameTypeBonus(m.type,s.fighters[0].type,s.typeChart)*affinity*roll/100));assert.equal(event.settledDamage,expected,m.name);assert.equal(event.hpLoss,expected);assert.equal(s.fighters[1].hp,9000-expected);}});
test('4000基础14935灵光闪原先制1覆盖速度劣势，10047重击不加先制时真实后手且都造成HP减少',()=>{for(const id of[10047,14935]){const s=make({move:id,index:-1,aStats:{spe:1},bStats:{spe:500}});turn(s,id,10127);assert.equal(s.turns[0].first,id===14935?0:1);assert.equal(s.fighters[0].soulState.resolvedPriority.value,id===14935?1:0);assert.ok(s.fighters[1].hp<9000);}});
