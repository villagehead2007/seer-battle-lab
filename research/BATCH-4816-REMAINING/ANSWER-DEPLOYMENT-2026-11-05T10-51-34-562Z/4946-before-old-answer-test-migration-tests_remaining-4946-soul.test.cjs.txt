'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,make,turn,timer,sum}=require('./helpers/remaining-4946.cjs'),S=require('../dist/shield-rules.js');
test('4946/2146回合初正级按种类附70固定，不按级数，非全雷配招不施麻痹',()=>{for(const level of[1,6]){const s=make({move:38536,index:-1,soul:true,loadout:[10006,38536,29455,29456,38538]});s.fighters[1].stages.atk=level;s.fighters[1].stages.spd=level;turn(s,38536,20309);assert.equal(sum(s,'fixed'),140);assert.equal(s.fighters[0].soulState.thunder2146.fixedKinds,2);assert.equal(s.fighters[1].ailments['麻痹'],undefined);}});
test('4946/2146全雷配招满体用攻真实麻痹，免异失败明确转待答吸取并回滚',()=>{const s=make({move:38536,index:-1,soul:true});s.fighters[1].stages.atk=1;turn(s,38536,20309);assert.ok(s.fighters[1].ailments['麻痹']);assert.equal(sum(s,'fixed'),70);const p=make({move:38536,index:-1,soul:true});p.fighters[1].stages.atk=1;p.fighters[1].counts.immune=1;assert.throws(()=>turn(p,38536,20309),/Q-4946-001/);assert.equal(p.round,1);assert.equal(p.fighters[1].counts.immune,1);});
test('4946/2146低于三分之一危急只一次魂印回复并清敌回合及普通盾，护罩保留；雷辰授予独立待答',()=>{const ability=E.SOULS.registry[2146].abilities,old=ability.gainFour;let requests=0;ability.gainFour=()=>requests++;try{for(const hp of[2999,3000]){const s=make({move:29455,index:-1,soul:true,hp});timer(s,0,'技能禁疗');S.refresh(s.fighters[1],300,'ordinary');S.refreshBarrier(s.fighters[1],400,'barrier');turn(s,29455,20117);assert.equal(s.fighters[0].hp,hp===2999?9000:3000);assert.equal(s.fighters[1].suitState.shield,hp===2999?0:300);assert.equal(s.fighters[1].suitState.barrier,400);assert.equal(s.fighters[0].soulState.thunder2146.rescueUsed,hp===2999);if(hp===2999){s.fighters[0].hp=1000;turn(s,29455,20117);assert.equal(s.fighters[0].hp,1000);}}assert.equal(requests,1);}finally{ability.gainFour=old;}const s=make({move:29455,index:-1,soul:true,hp:1000});assert.throws(()=>turn(s,29455,20117),/Q-4946-003/);assert.equal(s.fighters[0].hp,1000);});
test('4946/2146无强化攻击的30%减伤口径仍待答，不能静默放过；敌点亮雷辰转化亦明确阻止',()=>{const s=make({move:38536,index:-1,soul:true});timer(s,1,'额外攻击减伤',{fraction:.5});timer(s,1,'全伤害减半');assert.throws(()=>turn(s,38536,20309),/Q-4946-002/);assert.equal(s.round,1);const p=make({move:29455,index:-1,soul:true});p.fighters[1].soulState.lei1966={stars:[0],connections:[],last:0};assert.throws(()=>turn(p,29455,20309),/Q-4946-003/);assert.deepEqual(p.fighters[1].soulState.lei1966.stars,[0]);});

test('4946/2146单来源攻击减伤严格最多30%，20%保持、50%及80%与30%结果一致，无减伤可正常攻击',()=>{const damage=[];for(const fraction of[0,.2,.3,.5,.8]){const s=make({move:38536,index:-1,soul:true});if(fraction)timer(s,1,'额外攻击减伤',{fraction});turn(s,38536,20309);damage.push(sum(s));}assert.ok(damage[0]>damage[1]&&damage[1]>damage[2]);assert.equal(damage[2],damage[3]);assert.equal(damage[2],damage[4]);});
