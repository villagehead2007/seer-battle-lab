'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,make,turn,key,timer,sum}=require('./helpers/remaining-4951.cjs');
function withoutCapacity(run){const old=E.SOULS;E.SOULS={...old,run(e,c){if(e==='beforeUse'&&c.a.side===0&&c.a.soulId===2151&&['physical','special'].includes(c.m?.kind))return;old.run(e,c);}};try{return run();}finally{E.SOULS=old;}}
test('4951/2151临时上限当前HP与速度等级待答保持真实整回合回滚，独立容量公式零等级按60%面板',()=>{const s=make({move:10001,soul:true});assert.throws(()=>turn(s,10001,20309),/Q-4951-002/);assert.equal(s.round,1);assert.equal(s.fighters[0].hp,9000);const old=E.SOULS;let capacity;E.SOULS={...old,run(e,c){if(e==='beforeUse'&&c.a.side===0){capacity=old.registry[2151].abilities.capacity(c);return;}old.run(e,c);}};try{turn(s,10001,20309);}finally{E.SOULS=old;}assert.deepEqual(capacity,{speed:500,rate:60,amount:300});const x=make({move:10001,soul:true});x.fighters[0].stages.spe=1;assert.throws(()=>turn(x,10001,20309),/Q-4951-001/);});
test('4951/2151已损HP附粉60至120独立递增封顶，属性与药剂不递增，粉免仍执行记录',()=>withoutCapacity(()=>{const s=make({move:10001,soul:true,hp:8000,loadout:[10001,20017,29468,29469,38572]});for(const rate of[60,80,100,120,120]){s.fighters[1].hp=9000;turn(s,10001,20309);assert.equal(sum(s,'percent'),1000*rate/100);}const before=s.fighters[0].soulState.wind2151.damageRate;turn(s,20017,20309);turn(s,null,20309);assert.equal(s.fighters[0].soulState.wind2151.damageRate,before);const x=make({move:10001,soul:true,hp:8000});timer(x,1,'固定百分比免疫');turn(x,10001,20309);assert.equal(sum(x,'percent'),0);assert.equal(x.fighters[0].soulState.wind2151.damageRate,80);}));
test('4951/2151回合初满HP增红60至120独立封顶，后续失血不取消当回合增红，非满不增层',()=>withoutCapacity(()=>{const s=make({move:10001,soul:true}),b=make({move:10001});for(const rate of[60,80,100,120,120]){for(const x of[s,b]){x.fighters[0].hp=9000;x.fighters[1].hp=9000;x.rng=12345;}turn(s,10001,20309);turn(b,10001,20309);assert.ok(Math.abs(sum(s)/sum(b)-(1+rate/100))<.05);}const x=make({move:10001,soul:true,hp:8999});turn(x,10001,20309);assert.equal(x.fighters[0].soulState.wind2151.fullRate,60);assert.equal(x.fighters[0].soulState.wind2151.fullBonus,0);}));
module.exports={withoutCapacity};
