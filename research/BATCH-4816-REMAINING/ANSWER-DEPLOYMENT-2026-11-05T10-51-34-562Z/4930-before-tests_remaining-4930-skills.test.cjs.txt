'use strict';const test=require('node:test'),assert=require('node:assert/strict'),{E,source,make,turn,timer,key,sum,keys,mark,observe}=require('./helpers/remaining-4930-audit.cjs');
const row=id=>source.moves.flatMap(m=>m.effectInstances.map((e,index)=>({move:m.id,index,id:e.id}))).find(e=>e.id===id);
test('4930/2530衰弱命中与免异失败清自身异常，零红仍执行命中效果',()=>{for(const blocked of[false,true]){const s=make(row(2530));s.fighters[0].ailments['凝滞']=3;timer(s,1,'攻击伤害免疫');if(blocked)s.fighters[1].counts.immune=1;turn(s,s.auditMoveId,20309);assert.equal(sum(s),0);assert.equal(!!s.fighters[1].ailments['衰弱'],!blocked);assert.equal(!!s.fighters[0].ailments['凝滞'],!blocked);}});
test('4930/453消强成功臣服，强化保护失败不附；1700消弱成功免异且失败不授',()=>{for(const blocked of[false,true]){const s=make(row(453));s.fighters[1].stages.atk=2;if(blocked)timer(s,1,'强化保护');turn(s,s.auditMoveId,20309);assert.equal(s.fighters[1].stages.atk,blocked?2:0);assert.equal(!!s.fighters[1].ailments['臣服'],!blocked);}for(const weak of[false,true]){const s=make(row(1700));if(weak)s.fighters[0].stages.atk=-2;turn(s,s.auditMoveId,20309);assert.equal(s.fighters[0].stages.atk,0);assert.equal(s.fighters[0].counts.immune??0,weak?1:0);}});
test('4930/2533臣服失败回复全部PP两点，成功不补；2531按全场不同印记种类加红伤',()=>{for(const blocked of[false,true]){const s=make(row(2533));for(const k of s.fighters[0].moves)s.fighters[0].pp[k]=2;if(blocked)s.fighters[1].counts.immune=1;turn(s,s.auditMoveId,20309);assert.equal(s.fighters[0].pp[key(38483)],blocked?4:2);assert.equal(s.fighters[0].pp[key(29419)],blocked?3:1);}const damage=[];for(const count of[0,1,2]){const s=make(row(2531));if(count){mark(s,0,'金');mark(s,1,'金');}if(count===2)mark(s,1,'水');turn(s,s.auditMoveId,20309);damage.push(sum(s));}assert.ok(damage[1]>damage[0]);assert.ok(damage[2]>damage[1]);assert.ok(Math.abs(damage[1]/damage[0]-1.25)<.02);assert.ok(Math.abs(damage[2]/damage[0]-1.5)<.02);});
test('4930/2527明确整五剩余时长授实际固粉免疫，取整有分歧则阻塞回滚',()=>{const s=make(row(2527));mark(s,0,'金',5);mark(s,1,'水',5);turn(s,s.auditMoveId,10716);assert.equal(s.fighters[0].effects['帝皇印记固粉免疫'].end,2);assert.equal(sum(s,'fixed',1),0);assert.ok(sum(s,'attack',1)>0);const p=make(row(2527));mark(p,0,'金',3);mark(p,1,'水',3);assert.throws(()=>turn(p,p.auditMoveId,20117),/Q-4930-003/);assert.equal(p.round,1);});
test('4930/2171真实负级累计固定理论吸取，敌强化翻倍，伤害免疫不削减理论回复，禁疗单独处理',()=>{for(const mode of['normal','boost','immune','ban']){const s=make({...row(2171),hp:1000});s.fighters[0].stages.atk=-2;s.fighters[0].stages.def=-1;if(mode==='boost')s.fighters[1].stages.def=1;if(mode==='immune')timer(s,1,'固定百分比免疫');if(mode==='ban')timer(s,0,'技能禁疗');turn(s,s.auditMoveId,20309);const n=mode==='boost'?360:180;assert.equal(sum(s,'fixed'),mode==='immune'?0:n);assert.equal(s.fighters[0].hp,mode==='ban'?1000:1000+n);assert.equal(s.fighters[0].stages.atk,-2);}});
test('4930/668对手正级才追加先制，与881自身弱化先制叠加',()=>{for(const boost of[false,true]){const s=make(row(668));if(boost)s.fighters[1].stages.def=1;turn(s,s.auditMoveId,20309);assert.equal(s.fighters[0].soulState.resolvedPriority.value,boost?1:0);}const s=make({move:38484});s.fighters[0].stages.atk=-1;s.fighters[1].stages.def=1;turn(s,38484,20309);assert.equal(s.fighters[0].soulState.resolvedPriority.value,4);assert.equal(s.fighters[0].stages.atk,-1);});
