function observe(fn,run){const old=E.SOULS;E.SOULS={...old,run(e,c){old.run(e,c);fn(e,c);}};try{return run();}finally{E.SOULS=old;}}
function warm(o={}){const s=make({move:10086,soul:true,...o});turn(s);return s;}
