test('4329 1959消回合成功睡眠失败才本招攻击翻倍，未清除不触发且清除先于伤害',()=>{let base;for(const mode of ['none','sleep','immune','protected']){const s=make(pick(1959)[0]);if(mode!=='none')timer(s,1,'测试回合');if(mode==='immune')s.fighters[1].counts.immune=1;if(mode==='protected')timer(s,1,'protect');const o=turn(s);if(mode==='none')base=damage(o);assert.equal(!!o.fighters[1].ailments['睡眠'],mode==='sleep');assert.equal(!!o.fighters[1].effects['测试回合'],mode==='protected');assert.ok(Math.abs(damage(o)-base*(mode==='immune'?2:1))<=1);if(mode==='immune')assert.equal(o.fighters[1].counts.immune,0);}});
test('4329 1959失误不消回合不触发奖励，技能失效不施加睡眠',()=>{for(const mode of ['miss','invalid']){const s=make(pick(1959)[0],{hit:false});timer(s,1,'测试回合');if(mode==='miss'){s.moveDefinitions[key(s.auditMoveId)].alwaysHit=false;s.moveDefinitions[key(s.auditMoveId)].accuracy=0;}else timer(s,0,'skillBlock');const o=turn(s);assert.ok(o.fighters[1].effects['测试回合']);assert.equal(o.fighters[1].ailments['睡眠'],undefined);assert.equal(damage(o),0);}});
